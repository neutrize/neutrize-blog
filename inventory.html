ANTHROPIC API KEY — 納品書AI解析用
納品書写真をAIで解析するために必要です。セキュリティのため、APIキーはブラウザに保存せず Google Apps Scriptのスクリプトプロパティ にのみ保存し、サーバー側（GAS）からAnthropic APIを呼び出す構成にしています。
console.anthropic.com でキーを取得し、下のGASコードの手順4でスクリプトプロパティに設定してください。
⚠ ブラウザにAPIキーを直接入力・保存する方式は、開発者ツールやネットワーク通信からキーが漏洩する危険があるため廃止しました。
GOOGLE APPS SCRIPT — スプレッドシート自動保存
入庫データ・棚卸しデータ・商品マスタ・カテゴリの変更をすべてGoogleスプレッドシートに自動保存します。
下のGASコードをコピーして設定してください。
GASコード — スプレッドシートのツール→スクリプトエディタに貼り付けてください
const SHEET_ID = 'YOUR_SPREADSHEET_ID'; // ★ここを変更

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);

    // 納品書AI解析（APIキーはスクリプトプロパティのみで保持し、ブラウザには渡さない）
    if (data.type === 'analyze') {
      return handleAnalyze(data);
    }

    const ss = SpreadsheetApp.openById(SHEET_ID);

    if (data.type === 'delivery') {
      const sheet = ss.getSheetByName('入庫ログ') || ss.insertSheet('入庫ログ');
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(['日付','仕入先','商品名','カテゴリ','数量','単価','小計']);
      }
      data.items.forEach(item => {
        sheet.appendRow([
          data.date, data.supplier, item.name,
          item.category, item.qty, item.price, item.qty * item.price
        ]);
      });
    }

    if (data.type === 'stocktake') {
      const sheet = ss.getSheetByName('棚卸しログ') || ss.insertSheet('棚卸しログ');
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(['日時','商品名','カテゴリ','在庫数','適正量']);
      }
      data.items.forEach(item => {
        sheet.appendRow([
          new Date(), item.name, item.cat, item.stock, item.target
        ]);
      });
    }

    // 商品マスタ（追加・編集・削除のたびに現在の全件で上書き）
    if (data.type === 'master') {
      const sheet = ss.getSheetByName('商品マスタ') || ss.insertSheet('商品マスタ');
      sheet.clearContents();
      sheet.appendRow(['商品名','カテゴリ','適正量','単位','仕入先','現在庫']);
      data.items.forEach(item => {
        sheet.appendRow([
          item.name, item.cat, item.target, item.unit, item.supplier || '', item.stock
        ]);
      });
    }

    // カテゴリ一覧（追加・編集・削除のたびに現在の全件で上書き）
    if (data.type === 'category') {
      const sheet = ss.getSheetByName('カテゴリ') || ss.insertSheet('カテゴリ');
      sheet.clearContents();
      sheet.appendRow(['カテゴリ名']);
      data.categories.forEach(cat => sheet.appendRow([cat]));
    }

    return ContentService
      .createTextOutput(JSON.stringify({status:'ok'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error', message: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// 納品書写真をAnthropic APIへ中継（APIキーはここでのみ読み込む）
function handleAnalyze(data) {
  const apiKey = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_API_KEY');
  if (!apiKey) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error', message:'ANTHROPIC_API_KEYが未設定です。プロジェクトの設定 → スクリプトプロパティ で追加してください。'}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    contentType: 'application/json',
    headers: {
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01'
    },
    payload: JSON.stringify({
      model: 'claude-sonnet-4-6',
      max_tokens: 4096,
      temperature: 0,
      messages: [{
        role: 'user',
        content: [
          { type: 'image', source: { type: 'base64', media_type: data.mediaType, data: data.imageBase64 } },
          { type: 'text', text: data.prompt }
        ]
      }]
    }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  const body = response.getContentText();
  if (status !== 200) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error', message: 'APIエラー (' + status + '): ' + body}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService
    .createTextOutput(JSON.stringify({status:'ok', data: JSON.parse(body)}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({status:'ok', message:'NEUTRIZE在庫管理 GAS稼働中'}))
    .setMimeType(ContentService.MimeType.JSON);
}
手順
1. Googleスプレッドシートを新規作成
2. URLの /d/ の後のIDを SHEET_ID に貼り付け
3. ツール → スクリプトエディタ → 上のコードを貼り付けて保存
4. 左メニューの⚙ プロジェクトの設定 → スクリプトプロパティ → プロパティ名 ANTHROPIC_API_KEY に取得したAPIキーを追加して保存
5. デプロイ → 新しいデプロイ → 種類「ウェブアプリ」→ アクセス「全員」
6. 表示されたURLを上の入力欄に貼り付けて保存
